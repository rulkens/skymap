import { DOCS_SETTINGS } from '../../../data/docsSettings';
import { DOCS_SETTING_SIZES } from '../../../data/docsSettingSizes';

export const settings = (id) => {
  const rows = DOCS_SETTINGS.find((group) => group.id === id).rows;
  const switches = rows.every((row) => row.values === 'On or off');
  return {
    cols: 'detail',
    head: switches ? ['Switch', 'What it shows', 'At first'] : ['Setting', 'What it does', 'Values', 'At first'],
    rows: rows.map((row) =>
      switches
        ? [row.name, row.does, row.first]
        : [{ text: row.name, sub: row.control }, row.does, row.values, row.first],
    ),
  };
};

## Outside the Settings panel

These are not in the panel, and none of them is kept over a reload either.

| What | Where | Values | At first |
| --- | --- | --- | --- |
| Each panel, folded or open | A click on the panel’s title: Settings or Navigation | Open or folded | Open; folded on a screen narrower than 768 pixels |
| The interface, shown or hidden | The Tab key | Shown or hidden | Shown |
| Auto-rotate | The Auto-rotate button in the top bar | On or off | Off |
| The clock: its speed, and running or paused | The clock at the lower right; see [Time](/docs/guide/time/) | 15 speeds; running or paused | Running on the present at 1 s/s |
| Dome mode | `?dome` in the address | On or off | Off |
