import DocLink from '../../../components/DocLink.astro';
import { DATA_SOURCES } from '../../../data/dataSources';
import { GLOSSARY } from '../../../data/glossary';

export const Terms = ({ letter }) => (
  <dl>
    {GLOSSARY.filter((term) => term.term[0].toUpperCase() === letter).map((term) => {
      const source = DATA_SOURCES.find((row) => row.id === term.source);
      return (
        <>
          <dt id={term.id}>{term.term}</dt>{' '}
          <dd>
            {term.text}
            {source && (
              <>
                {' '}
                Source: <a href={source.href}>{source.name}</a>. Our use of it:{' '}
                <DocLink href="/science/#sources">The sources</DocLink>.
              </>
            )}
            {term.more && (
              <>
                {' '}
                More: <DocLink href={term.more.path}>{term.more.label}</DocLink>.
              </>
            )}
            {term.see && (
              <>
                {' '}
                See also{' '}
                {term.see.map((id, i) => (
                  <>
                    {i > 0 && ', '}
                    <a href={`#${id}`}>{GLOSSARY.find((other) => other.id === id).term}</a>
                  </>
                ))}
                .
              </>
            )}
          </dd>
        </>
      );
    })}
  </dl>
);

Each of the {GLOSSARY.length} terms has an address of its own: `/docs/reference/glossary/#redshift` opens at Redshift.

Three words have one fixed meaning across the site: [measured](#measured), [derived](#derived) and [drawn](#drawn). [The science behind skymap](/science/#bins) sorts everything the app shows under them.

## Units of distance

| Unit | In smaller units | In kilometres |
| --- | --- | --- |
| Light-second | | 299,792.458 |
| Light-minute | 60 light-seconds | about 18 million |
| Astronomical unit (AU) | 8.3 light-minutes | about 150 million |
| Light-day | 173 AU | about 25.9 thousand million |
| Light-year (ly) | 63,241 AU | about 9.46 million million |
| Parsec (pc) | 3.26 light-years, or 206,265 AU | about 30.9 million million |
| Kiloparsec (kpc) | 1,000 parsecs | |
| Megaparsec (Mpc) | 1,000 kiloparsecs, or 3.26 million light-years | |
| Gigaparsec (Gpc) | 1,000 megaparsecs, or 3.26 thousand million light-years | |
