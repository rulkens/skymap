import DocLink from '../../../components/DocLink.astro';
import { DATA_SOURCES } from '../../../data/dataSources';
import { GLOSSARY } from '../../../data/glossary';

export const Terms = ({ letter }) => (
  <dl>
    {GLOSSARY.filter((term) => term.term[0].toUpperCase() === letter).map((term) => {
      const source = DATA_SOURCES.find((row) => row.id === term.source);
      return (
        <>
          <dt id={term.id}>{term.term}</dt>
          <dd>
            {term.text}
            {source && (
              <>
                {' '}
                <a href={source.href}>Its own site or paper</a>, and its row under{' '}
                <DocLink href="/science/#sources">The sources</DocLink> on the Science page.
              </>
            )}
            {term.more && (
              <>
                {' '}
                More: <DocLink href={term.more.path}>{term.more.label}</DocLink>.
              </>
            )}
            {term.see && (
              <>
                {' '}
                See also{' '}
                {term.see.map((id, i) => (
                  <>
                    {i > 0 && ', '}
                    <a href={`#${id}`}>{GLOSSARY.find((other) => other.id === id).term}</a>
                  </>
                ))}
                .
              </>
            )}
          </dd>
        </>
      );
    })}
  </dl>
);

This page defines {GLOSSARY.length} terms in the order of the alphabet. Each has an address of its own: `/docs/reference/glossary/#redshift` opens at Redshift. The numbers in a definition come from the sources listed at the foot of the page.

Three words have one fixed meaning across the site: [measured](#measured), [derived](#derived) and [drawn](#drawn). The [Science page](/science/#bins) sorts everything the app shows under them.
