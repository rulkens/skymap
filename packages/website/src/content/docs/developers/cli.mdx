import { DOCS_COMMANDS, DOCS_COMMANDS_UNLISTED } from '../../../data/docsCommands';
import { REPO_BLOB } from '../../../data/siteIdentity';

export const commands = (id) => ({
  cols: 'detail',
  head: ['Command', 'What it does', 'Manual'],
  rows: DOCS_COMMANDS.find((group) => group.id === id).rows.map((row) => [
    { codes: row.scripts },
    row.does,
    { href: `${REPO_BLOB}/${row.manual}`, label: row.manual },
  ]),
});

export const Unlisted = () => (
  <ul>
    {DOCS_COMMANDS_UNLISTED.map((kind) => (
      <li>
        {kind.scripts.map((name, i) => (
          <>
            {i > 0 && ', '}
            <code>{name}</code>
          </>
        ))}
        : {kind.why}.
      </li>
    ))}
  </ul>
);
