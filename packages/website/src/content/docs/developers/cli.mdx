import pkg from '../../../../../../package.json';
import { DOCS_COMMANDS } from '../../../data/docsCommands';
import { REPO_BLOB } from '../../../data/siteIdentity';

export const NODE = pkg.engines.node.replace('>=', '');

export const members = (row) =>
  row.does.replace('<sources>', row.scripts.map((name) => name.slice(row.family.indexOf('<'))).join(', '));

export const commands = (id) => ({
  head: ['Command', 'What it does', 'Read'],
  rows: DOCS_COMMANDS.find((group) => group.id === id).rows.map((row) => [
    { codes: row.family ? [row.family] : row.scripts },
    row.family ? members(row) : row.does,
    { href: `${REPO_BLOB}/${row.manual}`, label: row.manual },
  ]),
});

Each is a script in the `package.json` at the root of the repository and needs Node {NODE} or later. Everything after a bare `--` is handed to the tool:

```
npm run shot -- 'focus=body-saturn' --url http://localhost:5173
```

## Pictures and films

`capture-featured`, `site:shots` and `site:loops` drive an app that is already running and take its address after `--url`, as `perf` does; so do `record-tour` and `record-clip` unless they are given `--serve`. `shot` starts an app itself when it is given no address.
